cmake_minimum_required(VERSION 3.20)
project(cutfem_thermoelastic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
option(CUTFEM_WITH_KOKKOS "Enable Kokkos volume assembly" OFF)
option(CUTFEM_WITH_HYPRE "Enable serial MPI hypre BoomerAMG-PCG" OFF)
add_library(cutfem src/mesh.cpp src/cut_layer.cpp src/assembly.cpp src/solvers.cpp src/experiments.cpp)
target_include_directories(cutfem PUBLIC include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(cutfem PRIVATE -Wall -Wextra -Wpedantic)
if(CUTFEM_WITH_KOKKOS)
  find_package(Kokkos 4.7 REQUIRED CONFIG)
  target_sources(cutfem PRIVATE src/kokkos_assembly.cpp)
  target_link_libraries(cutfem PUBLIC Kokkos::kokkos)
  target_compile_definitions(cutfem PUBLIC CUTFEM_WITH_KOKKOS=1)
endif()
if(CUTFEM_WITH_HYPRE)
  set(HYPRE_ROOT "/workspace/prototypes/p1-mfem-thinlayer/deps/hypre" CACHE PATH "hypre installation")
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_library(cutfem_hypre STATIC IMPORTED)
  set_target_properties(cutfem_hypre PROPERTIES IMPORTED_LOCATION "${HYPRE_ROOT}/lib/libHYPRE.a" INTERFACE_INCLUDE_DIRECTORIES "${HYPRE_ROOT}/include")
  target_link_libraries(cutfem PUBLIC cutfem_hypre MPI::MPI_CXX)
  # The installed archive uses GOT-indirect function-address pushes. Older system
  # linkers can mis-relax those into calls; preserve the original relocations.
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND UNIX AND NOT APPLE)
    target_link_options(cutfem PUBLIC "LINKER:--no-relax")
  endif()
  target_compile_definitions(cutfem PUBLIC CUTFEM_WITH_HYPRE=1)
endif()
if(CUTFEM_WITH_KOKKOS)
  add_executable(kokkos_bench apps/kokkos_bench.cpp)
  target_link_libraries(kokkos_bench PRIVATE cutfem)
endif()
add_executable(cutfem_proto apps/cutfem_proto.cpp tests/unit_tests.cpp)
target_link_libraries(cutfem_proto PRIVATE cutfem)
enable_testing()
add_test(NAME unit_tests COMMAND cutfem_proto unit_tests)
