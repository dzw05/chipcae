if(NOT TARGET chipcae::material)
    message(FATAL_ERROR "cte_contract: chipcae::material target missing")
endif()

# Reuse the existing fixtures without modifying the out-of-scope material tests.
# Track the source so fixture edits regenerate the header automatically.
set(_core_math "${PROJECT_SOURCE_DIR}/material/tests/core_math.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_core_math}")
file(READ "${_core_math}" _source)
set(_fixtures "#pragma once\n#include \"chipcae/material/cte_convert.hpp\"\nnamespace chipcae::material::test {\n")
foreach(_table sec tan)
    string(REGEX MATCH "CteTable ${_table}\\{[^;]+;" _declaration "${_source}")
    if(NOT _declaration)
        message(FATAL_ERROR "cte_contract: core_math ${_table} fixture missing")
    endif()
    string(APPEND _fixtures "inline const ${_declaration}\n")
endforeach()
foreach(_conversion tangent_to_secant secant_to_tangent)
    string(REGEX MATCH "${_conversion}\\([^,]+, ([0-9.]+), exact\\) == StatusCode::Ok && near\\(exact, ([0-9.e+-]+)\\)" _sample "${_source}")
    if(NOT _sample)
        message(FATAL_ERROR "cte_contract: core_math ${_conversion} sample missing")
    endif()
    string(APPEND _fixtures "inline constexpr Real ${_conversion}_temperature = ${CMAKE_MATCH_1};\ninline constexpr Real ${_conversion}_expected = ${CMAKE_MATCH_2};\n")
endforeach()
string(APPEND _fixtures "} // namespace chipcae::material::test\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core_math_cte_fixtures.hpp"
     CONTENT "${_fixtures}" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}")

chipcae_auto_tests(LIBS chipcae::material LABELS tierA PREFIX ctecontract_)
