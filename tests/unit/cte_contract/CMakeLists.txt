if(NOT TARGET chipcae::material)
    message(FATAL_ERROR "cte_contract: chipcae::material target missing")
endif()

# Keep the baseline read-only while deriving all canonical CTE data from it.
# Fail closed if its fixture syntax changes, rather than retaining stale literals.
set(_cte_source "${PROJECT_SOURCE_DIR}/material/tests/core_math.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_cte_source}")
file(READ "${_cte_source}" _cte_math)
function(cte_extract pattern output)
    string(REGEX MATCHALL "${pattern}" _matches "${_cte_math}")
    list(LENGTH _matches _count)
    if(NOT _count EQUAL 1)
        message(FATAL_ERROR "cte_contract: cannot uniquely extract ${output} from core_math.cpp")
    endif()
    string(REGEX MATCH "${pattern}" _match "${_cte_math}")
    set(${output} "${CMAKE_MATCH_1}" PARENT_SCOPE)
endfunction()
foreach(_table sec tan)
    cte_extract("CteTable[ \t\r\n]+${_table}([^{;]*\\{[^;]+\\})" ${_table}_initializer)
endforeach()
foreach(_direction secant_to_tangent tangent_to_secant)
    cte_extract("${_direction}\\([^,]+,[ \t\r\n]*([0-9.eE+-]+),[ \t\r\n]*exact\\)" ${_direction}_temperature)
    cte_extract("${_direction}\\([^;]+near\\(exact,[ \t\r\n]*([0-9.eE+-]+)\\)" ${_direction}_expected)
endforeach()
configure_file(cte_baseline.hpp.in cte_baseline.hpp @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}")

chipcae_auto_tests(LIBS chipcae::material LABELS tierA PREFIX ctecontract_)
