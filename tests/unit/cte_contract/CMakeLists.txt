if(NOT TARGET chipcae::material)
    message(FATAL_ERROR "cte_contract: chipcae::material target missing")
endif()

# Reuse the canonical test data without modifying material/tests/core_math.cpp.
set(cte_source "${PROJECT_SOURCE_DIR}/material/tests/core_math.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${cte_source}")
file(READ "${cte_source}" cte_source_text)
set(cte_fixture_text "#pragma once\n#include \"chipcae/material/cte_convert.hpp\"\nnamespace chipcae::material::test {\n")
foreach(table IN ITEMS sec tan)
    string(REGEX MATCH "CteTable ${table}\\{[^;]+;" declaration "${cte_source_text}")
    if(NOT declaration)
        message(FATAL_ERROR "cte_contract: canonical ${table} fixture not found")
    endif()
    string(APPEND cte_fixture_text "inline const ${declaration}\n")
endforeach()
foreach(conversion IN ITEMS tangent_to_secant secant_to_tangent)
    string(REGEX MATCH "${conversion}\\((sec|tan), ([0-9.]+), exact\\) == StatusCode::Ok && near\\(exact, ([0-9.eE+-]+)\\)" sample "${cte_source_text}")
    if(NOT sample)
        message(FATAL_ERROR "cte_contract: canonical ${conversion} sample not found")
    endif()
    string(APPEND cte_fixture_text
        "inline constexpr Real ${conversion}_temperature = ${CMAKE_MATCH_2};\n"
        "inline constexpr Real ${conversion}_expected = ${CMAKE_MATCH_3};\n")
endforeach()
string(APPEND cte_fixture_text "} // namespace chipcae::material::test\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cte_canonical_fixtures.hpp" "${cte_fixture_text}")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")

chipcae_auto_tests(LIBS chipcae::material LABELS tierA PREFIX ctecontract_)
