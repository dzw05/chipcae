if(NOT TARGET chipcae::material)
    message(FATAL_ERROR "cte_contract: chipcae::material target missing")
endif()

# Read, never modify, the canonical test. Track it so edits regenerate the header.
set(canonical_source "${PROJECT_SOURCE_DIR}/material/tests/core_math.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${canonical_source}")
file(READ "${canonical_source}" canonical_text)
set(generated "// Generated from material/tests/core_math.cpp; do not edit.\n#pragma once\n#include <chipcae/material/cte_convert.hpp>\nnamespace chipcae::material::test {\n")
foreach(table IN ITEMS sec tan)
    string(REGEX MATCHALL "CteTable[ \t\r\n]+${table}[ \t\r\n]*\\{[^;]+" matches "${canonical_text}")
    list(LENGTH matches count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "cte_contract: cannot uniquely extract canonical ${table} table")
    endif()
    string(APPEND generated "inline const ${matches};\n")
endforeach()
foreach(conversion IN ITEMS secant_to_tangent tangent_to_secant)
    if(conversion STREQUAL "secant_to_tangent")
        set(table sec)
    else()
        set(table tan)
    endif()
    string(REGEX MATCHALL "${conversion}\\(${table},[ \t\r\n]*[0-9.]+,[ \t\r\n]*exact\\)[ \t\r\n]*==[ \t\r\n]*StatusCode::Ok[ \t\r\n]*&&[ \t\r\n]*near\\(exact,[ \t\r\n]*[0-9.eE+-]+\\)" matches "${canonical_text}")
    list(LENGTH matches count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "cte_contract: cannot uniquely extract canonical ${conversion} sample")
    endif()
    string(REGEX MATCH "${conversion}\\(${table},[ \t\r\n]*([0-9.]+)," unused "${matches}")
    string(APPEND generated "inline constexpr Real ${conversion}_temperature = ${CMAKE_MATCH_1};\n")
    string(REGEX MATCH "near\\(exact,[ \t\r\n]*([0-9.eE+-]+)\\)" unused "${matches}")
    string(APPEND generated "inline constexpr Real ${conversion}_expected = ${CMAKE_MATCH_1};\n")
endforeach()
string(APPEND generated "} // namespace chipcae::material::test\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/canonical_cte_fixtures.hpp" CONTENT "${generated}" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}")

chipcae_auto_tests(LIBS chipcae::material LABELS tierA PREFIX ctecontract_)
