find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(GLOB _cpp_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp")
file(GLOB _python_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.py")
foreach(_file IN LISTS _cpp_tests _python_tests)
    get_filename_component(_name "${_file}" NAME_WE)
    string(REGEX REPLACE "^test_" "" _name "${_name}")
    if(_name MATCHES "^occt_" AND NOT CHIPCAE_ENABLE_OCCT)
        continue()
    endif()
    if(_file MATCHES "[.]cpp$")
        add_executable(chipcae_test_${_name} "${_file}")
        target_link_libraries(chipcae_test_${_name} PRIVATE chipcae::geometry chipcae_mesh_cut)
        chipcae_set_warnings(chipcae_test_${_name})
        add_test(NAME ${_name} COMMAND chipcae_test_${_name})
    else()
        add_test(NAME ${_name} COMMAND ${Python3_EXECUTABLE} "${_file}")
        set_tests_properties(${_name} PROPERTIES SKIP_RETURN_CODE 77)
    endif()
    set_tests_properties(${_name} PROPERTIES LABELS "tierA" TIMEOUT 60)
    if(_name MATCHES "^occt_")
        set_tests_properties(${_name} PROPERTIES LABELS "occt")
    endif()
endforeach()
