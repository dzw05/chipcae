  # Defer until all targets exist. Resolve aliases before evaluating both link
  # properties; TARGET_GENEX_EVAL handles the active configuration, LINK_ONLY,
  # and nested conditional expressions rather than stripping them as text.
  function(i5_collect_directory directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    get_property(imported DIRECTORY "${directory}" PROPERTY IMPORTED_TARGETS)
    set_property(GLOBAL APPEND PROPERTY i5_targets ${targets} ${imported})
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
      i5_collect_directory("${child}")
    endforeach()
  endfunction()
  function(i5_generate_link_graph)
    i5_collect_directory("${PROJECT_SOURCE_DIR}")
    get_property(queue GLOBAL PROPERTY i5_targets)
    set(seen "")
    set(graph "")
    while(queue)
      list(POP_FRONT queue t)
      if(t IN_LIST seen OR NOT TARGET "${t}")
        continue()
      endif()
      list(APPEND seen "${t}")
      get_target_property(alias "${t}" ALIASED_TARGET)
      if(alias)
        string(APPEND graph "set(links_${t} ${alias})\n")
        list(APPEND queue "${alias}")
        continue()
      endif()
      get_target_property(type "${t}" TYPE)
      if(type STREQUAL "UTILITY")
        continue()
      endif()
      foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
        get_target_property(raw "${t}" "${property}")
        # Discover target spellings inside generator expressions, including aliases.
        string(REGEX MATCHALL "[A-Za-z_][A-Za-z0-9_.:+-]*" tokens "${raw}")
        foreach(token IN LISTS tokens)
          if(TARGET "${token}")
            list(APPEND queue "${token}")
          endif()
        endforeach()
      endforeach()
      string(APPEND graph "set(links_${t} [==[$<TARGET_GENEX_EVAL:${t},$<TARGET_PROPERTY:${t},LINK_LIBRARIES>>;$<TARGET_GENEX_EVAL:${t},$<TARGET_PROPERTY:${t},INTERFACE_LINK_LIBRARIES>>]==])\n")
    endwhile()
    file(GENERATE OUTPUT "${PROJECT_BINARY_DIR}/i5-links-$<CONFIG>.cmake" CONTENT "${graph}")
  endfunction()
  cmake_language(DEFER DIRECTORY "${PROJECT_SOURCE_DIR}" CALL i5_generate_link_graph)
